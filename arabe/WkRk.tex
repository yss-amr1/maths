\ex{Applications}{WkRk}{Ex. 3}{9}
On considère l'application $f$ de $I=\,]2,+\infty[$ vers $I$ définie par $f(x)=\dfrac{2x}{\sqrt{x^2-4}}$.
\begin{enumerate}
\item Montrer que $f$ est injective.
\item Montrer que $(\forall x\in I)\ f(x)\in I$.
\item Calculer $(f\circ f)(x)$ pour tout $x$ de $I$, et en déduire que $f$ est une bijection et déterminer sa bijection réciproque.
\end{enumerate}
\fin
\ex{Applications}{WkRk}{Ex. 6}{9}
Soit $f$ l'application de $[1,+\infty[$ vers $\mathbb{R}^+$ définie par $f(x)=\sqrt{x+1}-\sqrt{x-1}$.
\begin{enumerate}
\item Résoudre dans $[1,+\infty[$ l'équation $f(x)=0$. Que peut-on en déduire ?
\item Montrer que $(\forall x\in[1,+\infty[)\ 0<f(x)\leq\sqrt2$.
\item Montrer que $f$ est injective.
\item \begin{enumerate}
\item Vérifier que $\sqrt{x+1}+\sqrt{x-1}=\dfrac{2}{f(x)}$.
\item Montrer que $f$ est une bijection de $[1,+\infty[$ vers $]0,\sqrt2]$ et déterminer sa bijection réciproque.
\end{enumerate}
\end{enumerate}
\fin
\ex{Applications}{WkRk}{Ex. 7}{9}
On considère l'application $f$ définie sur $\mathbb{R}$ par $f(x)=\sqrt{4+x^2}-|x|$.
\begin{enumerate}
\item $f$ est-elle injective ?
\item Montrer que $(\forall x\in\mathbb{R})\ f(x)>0$. $f$ est-elle surjective ?
\item Soit $g$ la restriction de $f$ à l'intervalle $\mathbb{R}^+$.
\begin{enumerate}
\item Montrer que $g$ est injective.
\item Montrer que $g$ est une bijection de $\mathbb{R}^+$ vers $]0,2]$ et déterminer sa bijection réciproque.
\end{enumerate}
\end{enumerate}
\fin
