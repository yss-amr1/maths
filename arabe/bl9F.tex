\ex{Ensembles}{bl9F}{Ex. 1}{5}
On considère les ensembles $A=\{x\in\mathbb{R}\ /\ |3x-2|\leq6\}$ et $B=\left\{x\in\mathbb{Z}\ /\ \dfrac{12}{x}\in\mathbb{Z}\right\}$.
Déterminer $A$ et $B$ en extension, puis déterminer $A\cup B$ et $A\cap B$.
\fin
\ex{Ensembles}{bl9F}{Ex. 2}{6}
On considère les ensembles $A=\left\{x\in\mathbb{R}\ /\ \dfrac{x-2}{x+2}\leq3\right\}$ et $B=\{x\in\mathbb{Z}\ /\ |x+1|+|x-2|=3\}$.
Déterminer $A$ et $B$ en extension, puis déterminer $A-B$ et $A\cap B$.
\fin
\ex{Ensembles}{bl9F}{Ex. 3}{6}
Déterminer les ensembles $A$ et $B$ sachant que : $A\cap B=\{1,2,3,4\}$, $A\cup B=\{1,2,3,4,5,6\}$, $5\notin B\cap\overline A$ et $6\notin A\cap\overline B$.
\fin
\ex{Ensembles}{bl9F}{Ex. 9}{9}
\textbf{I.} On considère l'ensemble $A=\left\{\sqrt{x+2}-\sqrt x\ /\ x\in\mathbb{R}^{+*}\right\}$.
\begin{enumerate}
\item Montrer que $A\subset\,]0,\sqrt2[$.
\item A-t-on $A=\,]0,\sqrt2[$ ?
\end{enumerate}
\textbf{II.} On pose $A=\left\{x=a+\dfrac1a\ /\ a\in\mathbb{R}^{+*}\right\}$.
\begin{enumerate}
\item Montrer que $A\subset[2,+\infty[$.
\item Soit $b$ un élément de $[2,+\infty[$. Calculer $\dfrac{b+\sqrt{b^2-4}}{2}+\dfrac{2}{b+\sqrt{b^2-4}}$.
\item A-t-on $A=[2,+\infty[$ ?
\end{enumerate}
\fin
